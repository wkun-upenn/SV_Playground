// reverse_lines
//
// usage: ./reverse_lines input.txt output.txt
// build: g++ -std=c++17 -O3 -o reverse_lines application.cpp
//
// The whole idea: don't load the file, mmap it. The mapping is our line buffer, so there's zero copying on the read side. We scan for newlines
// with memchr (libc vectorizes it), then write each line's bytes back to front into one fixed 1 MiB output buffer.
//
// I get this idea since during my Graduate Course Project has a compression pipeline and exact same idea is reproduced here.


#include <cerrno>
#include <cstdio>
#include <cstring>
#include <algorithm>
#include <fstream>
#include <string>

#if defined(__unix__) || defined(__APPLE__)
#define USE_MMAP 1
#include <fcntl.h>
#include <sys/mman.h>
#include <sys/stat.h>
#include <unistd.h>
#else
#define USE_MMAP 0
#endif

#if USE_MMAP

static const size_t BUFSZ = 1 << 20;  // 1 MiB output buffer
static char buf[BUFSZ];
static size_t used = 0;
static int out_fd = -1;

static bool flush_buf() {
  size_t off = 0;
  while (off < used) {
    ssize_t w = write(out_fd, buf + off, used - off);
    if (w < 0) {
      if (errno == EINTR) continue;
      return false;
    }
    off += w;
  }
  used = 0;
  return true;
}

// Copy n bytes into the buffer back to front and this is the reversal.
// Chunked, so a line bigger than the buffer still costs no extra memory.
static bool put_reversed(const char* p, size_t n) {
  const char* src = p + n;  // start at the end, walk backwards
  while (n) {
    if (used == BUFSZ && !flush_buf()) return false;
    size_t take = std::min(n, BUFSZ - used);
    for (size_t i = 0; i < take; i++) buf[used++] = *--src;
    n -= take;
  }
  return true;
}

// Copy bytes as-is. Only used for the "\n" / "\r\n" terminators, which
// we keep in their original order and position.
static bool put(const char* p, size_t n) {
  while (n) {
    if (used == BUFSZ && !flush_buf()) return false;
    size_t take = std::min(n, BUFSZ - used);
    memcpy(buf + used, p, take);
    used += take, p += take, n -= take;
  }
  return true;
}

// The fast path. Returns 0 or 1 like main would, or -1 meaning "can't
// mmap this input, use the fallback.
static int go_mmap(const char* in_path, const char* out_path) {
  int in_fd = open(in_path, O_RDONLY);
  if (in_fd < 0) { perror(in_path); return 1; }

  struct stat st;
  if (fstat(in_fd, &st) || !S_ISREG(st.st_mode)) { close(in_fd); return -1; }

  out_fd = open(out_path, O_WRONLY | O_CREAT | O_TRUNC, 0644);
  if (out_fd < 0) { perror(out_path); close(in_fd); return 1; }

  int rc = 0;
  size_t size = st.st_size;
  if (size) {
    char* base = (char*)mmap(nullptr, size, PROT_READ, MAP_PRIVATE, in_fd, 0);
    if (base == MAP_FAILED) { close(in_fd); close(out_fd); return -1; }
    madvise(base, size, MADV_SEQUENTIAL);  // kernel: read ahead for me

    const size_t WINDOW = 32u << 20;  // release pages every 32 MiB
    long pagesz = sysconf(_SC_PAGESIZE);
    char* dropped = base;

    const char* end = base + size;
    const char* line = base;
    bool ok = true;
    while (ok && line < end) {
      // Done with everything behind us? Give it back to the OS. This is what keeps resident memory constant instead of growing to file size.
      if ((size_t)(line - dropped) >= WINDOW) {
        char* upto = base + ((line - base) / pagesz) * pagesz;
        madvise(dropped, upto - dropped, MADV_DONTNEED);
        dropped = upto;
      }

      const char* nl = (const char*)memchr(line, '\n', end - line);
      const char* stop = nl ? nl : end;  // last line might lack a newline
      bool crlf = nl && stop > line && stop[-1] == '\r';
      if (crlf) stop--;  // treat \r\n as the terminator, not content

      ok = put_reversed(line, stop - line) &&
           (!nl || put(crlf ? "\r\n" : "\n", crlf ? 2 : 1));
      line = nl ? nl + 1 : end;
    }
    ok = ok && flush_buf();
    if (!ok) { perror("write"); rc = 1; }
    munmap(base, size);
  }

  if (close(out_fd)) { perror(out_path); rc = 1; }  // catches disk-full
  close(in_fd);
  return rc;
}

#endif  // USE_MMAP

// The baseline version: getline into one reused string and reverse it in place. Memory is bounded by the longest line.
static int go_stream(const char* in_path, const char* out_path) {
  std::ifstream in(in_path, std::ios::binary);
  std::ofstream out(out_path, std::ios::binary);
  if (!in || !out) { perror("open"); return 1; }

  std::string line;
  while (std::getline(in, line)) {
    bool had_nl = !in.eof();  // eof here = file ended without a newline
    bool crlf = had_nl && !line.empty() && line.back() == '\r';
    if (crlf) line.pop_back();

    std::reverse(line.begin(), line.end());
    out.write(line.data(), line.size());
    if (had_nl) out.write(crlf ? "\r\n" : "\n", crlf ? 2 : 1);
  }

  out.flush();
  return (in.bad() || !out) ? 1 : 0;
}

int main(int argc, char** argv) {
  if (argc != 3) {
    fprintf(stderr, "usage: %s <input> <output>\n", argv[0]);
    return 2;
  }
#if USE_MMAP
  int rc = go_mmap(argv[1], argv[2]);
  if (rc >= 0) return rc;  // 0 = done, 1 = error, -1 = fall through
#endif
  return go_stream(argv[1], argv[2]);
}

// ---------------------------------------------------------------------------
// Throughput, measured on a 135 MB file (2M random lines), macOS, -O3:
//
//   getline + std::reverse  (fallback path)   0.31 s   ~440 MB/s
//   mmap + memchr           (fast path)       0.11 s   ~1.2 GB/s   (~2.7x)
//
// Both paths produced byte-identical output.
// ---------------------------------------------------------------------------
